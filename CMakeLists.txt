cmake_minimum_required(VERSION 3.21)
project(WasteSort VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network Test)
qt_standard_project_setup(REQUIRES 6.8)
add_library(wastesort_core STATIC src/controller.cpp src/controller.h)
target_link_libraries(wastesort_core PUBLIC Qt6::Core Qt6::Network)
target_include_directories(wastesort_core PUBLIC src)
qt_add_executable(WasteSort src/main.cpp)
if(WIN32)
    target_sources(WasteSort PRIVATE assets/app.rc)
    target_include_directories(WasteSort PRIVATE assets)
endif()
qt_add_resources(WasteSort app_icons PREFIX "/" FILES assets/guixu.png)
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(WasteSort URI WasteSort VERSION 1.0
    QML_FILES qml/Main.qml qml/Theme.qml qml/UiText.qml qml/UiButton.qml
    qml/Panel.qml qml/Field.qml qml/LineIcon.qml qml/BinArt.qml
    qml/LoginPage.qml qml/Dashboard.qml qml/LogPage.qml qml/SettingsPage.qml)
target_link_libraries(WasteSort PRIVATE wastesort_core Qt6::Quick Qt6::QuickControls2)
set_target_properties(WasteSort PROPERTIES WIN32_EXECUTABLE TRUE)
enable_testing()
qt_add_executable(controller_tests tests/controller_tests.cpp)
target_link_libraries(controller_tests PRIVATE wastesort_core Qt6::Test)
add_test(NAME controller_tests COMMAND controller_tests)
